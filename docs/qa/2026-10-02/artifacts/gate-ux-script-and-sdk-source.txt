10: async function layout(page) {
11:   return page.evaluate(() => ({
12:     viewport: window.innerWidth,
13:     documentWidth: document.documentElement.scrollWidth,
14:     headings: Array.from(document.querySelectorAll('h1,h2,h3')).map(e => ({tag: e.tagName, text: e.textContent.trim()})),
15:     landmarks: Array.from(document.querySelectorAll('main,nav,header,footer')).map(e => ({tag: e.tagName, name: e.getAttribute('aria-label')})),
16:     overflow: Array.from(document.querySelectorAll('main,nav,button,input,a,h1,h2,h3,pre,table')).filter(e => {
17:       const r = e.getBoundingClientRect();
18:       return r.width > 0 && (r.right > window.innerWidth + 1 || r.left < -1);
19:     }).map(e => ({tag: e.tagName, id: e.id, cls: e.className, text: e.textContent.trim().slice(0,80), width: e.getBoundingClientRect().width})),
20:   }));
21: }
