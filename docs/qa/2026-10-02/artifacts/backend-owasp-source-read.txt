FILE app/middleware/request_body_limit.py
1: """Bound the size of inbound HTTP request bodies.
2: 
3: The governed MCP invoke path accepts caller-supplied tool arguments, which are
4: parsed, hashed, and persisted. Without a ceiling, one request can force the
5: gateway to buffer and process an arbitrarily large payload before any permit,
6: wallet, or rate-limit decision has a chance to reject it — the work is done on
7: the attacker's terms and charged to nobody.
8: 
9: Two MCP transports already cap their own bodies (``mcp_public`` at 256 KiB,
10: ``partner_mcp`` at 64 KiB), but both are opt-in surfaces. This middleware puts
11: the same floor under every route, including ``/mcp/messages``, which is always
12: mounted and is the product.
13: 
14: Enforcement is two-stage. A declared ``Content-Length`` over the limit is
15: refused without reading a byte, so the common case costs nothing. Otherwise the
16: body is drained up to the limit and replayed to the application; a stream that
17: exceeds the limit mid-flight is refused at the moment it does. Buffering is
18: bounded by the limit itself, so the check cannot become the exhaustion it
19: prevents.
20: """
21: 
22: from __future__ import annotations
23: 
24: from collections import deque
25: 
26: from starlette.datastructures import Headers
27: from starlette.responses import JSONResponse
28: from starlette.types import ASGIApp, Message, Receive, Scope, Send
29: 
30: # Methods that carry a body. A GET has none to bound, and passing it straight
31: # through keeps the read side of the API free of buffering.
32: BODY_METHODS = frozenset({"POST", "PUT", "PATCH", "DELETE"})
33: 
34: 
35: def _too_large(limit: int) -> JSONResponse:
36:     """413 in the API's error shape, naming the limit so a caller can adapt."""
37:     return JSONResponse(
38:         status_code=413,
39:         content={
40:             "detail": f"Request body exceeds the {limit}-byte limit.",
41:             "max_request_body_bytes": limit,
42:         },
43:     )
44: 
45: 
46: class RequestBodyLimitMiddleware:
47:     """Reject request bodies larger than ``max_body_size`` with a 413."""
48: 
49:     def __init__(self, app: ASGIApp, max_body_size: int) -> None:
50:         self.app = app
51:         self.max_body_size = max_body_size
52: 
53:     async def __call__(self, scope: Scope, receive: Receive, send: Send) -> None:
54:         if scope["type"] != "http" or scope.get("method") not in BODY_METHODS:
55:             await self.app(scope, receive, send)
56:             return
57: 
58:         declared = Headers(scope=scope).get("content-length")
59:         if declared is not None:
60:             try:
61:                 declared_size = int(declared)
62:             except ValueError:
63:                 # A malformed header is not proof of an oversized body; fall
64:                 # through to the streaming check, which measures the real one.
65:                 pass
66:             else:
67:                 if declared_size > self.max_body_size:
68:                     await _too_large(self.max_body_size)(scope, receive, send)
69:                     return
70: 
71:         # A truthful Content-Length is not guaranteed (chunked encoding omits
72:         # it, a hostile client can understate it), so measure what arrives.
73:         buffered = bytearray()
74:         saw_body = False
75:         complete = False
76:         trailing: Message | None = None
77:         while True:
78:             message = await receive()
79:             if message["type"] != "http.request":
80:                 # A disconnect (or anything else) ends the body; hand it on so
81:                 # the application observes the same event we did.
82:                 trailing = message
83:                 break
84: 
85:             saw_body = True
86:             buffered.extend(message.get("body", b""))
87:             if len(buffered) > self.max_body_size:
88:                 await _too_large(self.max_body_size)(scope, receive, send)
89:                 return
90:             if not message.get("more_body", False):
91:                 complete = True
92:                 break
93: 
94:         replayed: deque[Message] = deque()
95:         if saw_body:
96:             replayed.append(
97:                 {
98:                     "type": "http.request",
99:                     "body": bytes(buffered),
100:                     "more_body": not complete,
101:                 }
102:             )
103:         if trailing is not None:
104:             replayed.append(trailing)
105: 
106:         async def replay() -> Message:
107:             if replayed:
108:                 return replayed.popleft()
109:             return await receive()
110: 
111:         await self.app(scope, replay, send)
FILE app/core/url_guard.py
1: """Outbound-URL guard for agent-driven navigation and proxied requests.
2: 
3: Agent-supplied URLs reach two genuinely dangerous sinks: the AWI Playwright
4: bridge (a real headless browser that will happily open ``file://`` and
5: intranet URLs) and the behavioral sandbox's HTTP proxy mode (a raw
6: ``aiohttp`` request). Both run inside our infrastructure, so an unchecked
7: URL is an SSRF primitive against link-local metadata services, RFC1918
8: networks, and the host filesystem.
9: 
10: ``check_outbound_url`` returns ``None`` when the URL is safe to fetch and a
11: short machine-readable reason string when it must be blocked. Scheme and
12: literal-address checks are unconditional; hostname checks resolve DNS and
13: block names that resolve to non-global addresses. Resolution failure fails
14: closed (``dns_resolution_failed``): the guard cannot vouch for a name it
15: could not resolve, and the fetching client (a headless browser with its own
16: resolver, or a later retry) may well resolve it — possibly to an intranet
17: address. This matches the upstream MCP URL guard. Tests that exercise
18: public hostnames pin ``_resolve_host`` instead of relying on live DNS. This
19: is a pre-flight guard, not a substitute for network-level egress policy,
20: and it intentionally does not try to defeat DNS rebinding.
21: 
22: Setting ``ALLOW_PRIVATE_NETWORK_TARGETS=true`` (local development against
23: mock servers) skips only the private-address checks; non-http(s) schemes
24: such as ``file://`` and ``javascript:`` are never allowed.
25: """
26: 
27: from __future__ import annotations
28: 
29: import asyncio
30: import ipaddress
31: import socket
32: from urllib.parse import urlsplit
33: 
34: from app.core.config import get_settings
35: 
36: _ALLOWED_SCHEMES = {"http", "https"}
37: _BLOCKED_HOSTNAMES = {"localhost", "metadata.google.internal"}
38: _BLOCKED_SUFFIXES = (".localhost", ".local", ".internal")
39: 
40: 
41: def _address_blocked(address: str) -> bool:
42:     try:
43:         ip = ipaddress.ip_address(address)
44:     except ValueError:
45:         return False
46:     # is_global is False for loopback, RFC1918/ULA, link-local (including
47:     # 169.254.169.254 metadata), CGNAT shared space, multicast, reserved,
48:     # and unspecified addresses.
49:     return not ip.is_global
50: 
51: 
52: async def _resolve_host(host: str):
53:     """Resolve a hostname through the event loop; split out for deterministic tests."""
54: 
55:     return await asyncio.get_running_loop().getaddrinfo(
56:         host, None, type=socket.SOCK_STREAM
57:     )
58: 
59: 
60: async def check_outbound_url(url: str) -> str | None:
61:     """Return a block reason for an agent-supplied outbound URL, or None."""
62:     try:
63:         parts = urlsplit(url)
64:     except ValueError:
65:         return "unparseable_url"
66: 
67:     if parts.scheme.lower() not in _ALLOWED_SCHEMES:
68:         return "scheme_not_allowed"
69: 
70:     host = parts.hostname
71:     if not host:
72:         return "missing_host"
73: 
74:     if get_settings().ALLOW_PRIVATE_NETWORK_TARGETS:
75:         return None
76: 
77:     host = host.strip("[]").rstrip(".").lower()
78:     if host in _BLOCKED_HOSTNAMES or host.endswith(_BLOCKED_SUFFIXES):
79:         return "private_host_blocked"
80: 
81:     if _address_blocked(host):
82:         return "private_address_blocked"
83: 
84:     try:
85:         ipaddress.ip_address(host)
86:     except ValueError:
87:         # A hostname, not a literal address: resolve it and require every
88:         # resolved address to be globally routable.
89:         try:
90:             infos = await _resolve_host(host)
91:         except (OSError, UnicodeError):
92:             # socket.gaierror is an OSError; UnicodeError covers names the
93:             # IDNA codec rejects (e.g. an over-long label). Fail closed.
94:             return "dns_resolution_failed"
95:         for info in infos:
96:             if _address_blocked(str(info[4][0])):
97:                 return "private_address_blocked"
98: 
99:     return None
