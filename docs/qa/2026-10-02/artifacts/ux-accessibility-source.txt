22:   function update() {
23:     next.hidden = true;
24:     alternative.hidden = true;
25:     guidance.textContent = "";
26:     for (const field of fields) {
27:       const invalid = field.validity.badInput ||
28:         (field.value !== "" && (!field.checkValidity() ||
29:           !Number.isFinite(field.valueAsNumber)));
30:       field.setAttribute("aria-invalid", String(invalid));
31:     }
32:     const invalid = fields.find((field) => field.getAttribute("aria-invalid") === "true");
33:     if (invalid) {
34:       summary.textContent = "Check your inputs: use non-negative numbers, whole actions, and percentages from 0 to 100.";
35:       return;
36:     }
37:     if (fields.some((field) => field.required && field.value === "")) {
38:       summary.textContent = "Enter the four workflow assumptions to estimate avoided loss.";
39:       return;
40:     }
41: 
42:     const [actions, rate, loss, reduction, cost] = fields.map((field) => field.valueAsNumber);
43:     const avoidedDuplicates = actions * (rate / 100) * (reduction / 100);
44:     const avoidedLoss = avoidedDuplicates * loss;
45:     const hasCost = fields[4].value !== "";
46:     const threshold = hasCost && avoidedDuplicates > 0 ? cost / avoidedDuplicates : 0;
47:     if (![avoidedDuplicates, avoidedLoss, threshold].every(Number.isFinite)) {
48:       summary.textContent = "These values are too large to calculate. Use smaller estimates.";
49:       return;
50:     }
51: 
52:     summary.textContent = "Estimated avoided duplicate loss: " + money.format(avoidedLoss) + " / month.";
53:     if (avoidedDuplicates === 0) {
54:       guidance.textContent = "These assumptions imply no avoided duplicates, so duplicate prevention alone cannot justify an added cost.";
55:       alternative.hidden = false;
56:     } else if (!hasCost) {
57:       guidance.textContent = "Economic fit is still unknown. Add your total monthly cost to compare; a service quote and operating scope are needed before deciding.";
58:     } else {
59:       const comparison = " Break-even loss per duplicate: " + money.format(threshold) + ".";
60:       // Use the same decimal rounding as the displayed currency amounts.
61:       const displayedLoss = Number(roundedAmount.format(avoidedLoss));
62:       const displayedCost = Number(roundedAmount.format(cost));
63:       if (displayedLoss > displayedCost) {
64:         guidance.textContent = "Estimated avoided loss exceeds total monthly cost by " + money.format(displayedLoss - displayedCost) + "." + comparison + " This is a reason to validate the assumptions in a paid pilot, not evidence of savings.";
65:         next.hidden = false;
66:       } else {
67:         guidance.textContent = "Estimated avoided loss does not exceed total monthly cost." + comparison + " Start with simpler reliability or upstream idempotency controls. Any separate operational benefit needs its own evidence.";
68:         alternative.hidden = false;
69:       }
70:     }
71:   }
