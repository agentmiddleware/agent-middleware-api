67: def canonical_json(
68:     payload: dict[str, Any],
69:     *,
70:     ensure_ascii: bool = True,
71:     allow_nan: bool = True,
72: ) -> str:
73:     """Serialize a payload into stable JSON before hashing or signing."""
74: 
75:     def normalize(value: Any) -> Any:
76:         if isinstance(value, Decimal):
77:             normalized = value.normalize()
78:             if normalized == normalized.to_integral():
79:                 return format(normalized, "f")
80:             return format(normalized, "f")
81:         if isinstance(value, datetime):
82:             if value.tzinfo is None:
83:                 value = value.replace(tzinfo=timezone.utc)
84:             return value.astimezone(timezone.utc).isoformat()
85:         if isinstance(value, dict):
86:             return {str(k): normalize(v) for k, v in sorted(value.items())}
87:         if isinstance(value, list):
88:             return [normalize(v) for v in value]
89:         return value
