567:def add_cors_middleware(application: FastAPI, origins: list[str]) -> None:
568:    """Attach CORS, enabling credentials only for an explicit allowlist.
569:
570:    Never pair credentialed CORS with a wildcard origin: with
571:    ``allow_origins=["*"]`` and ``allow_credentials=True``, Starlette echoes the
572:    caller's ``Origin`` back and still returns
573:    ``Access-Control-Allow-Credentials: true``, letting any website make
574:    credentialed cross-origin reads against the trust plane. A wildcard therefore
575:    serves ``Access-Control-Allow-Origin: *`` with credentials disabled.
576:    """
577:    allow_all_origins = "*" in origins
578:    application.add_middleware(
579:        CORSMiddleware,
580:        allow_origins=origins,
581:        allow_credentials=not allow_all_origins,
582:        allow_methods=["*"],
583:        allow_headers=["*"],
