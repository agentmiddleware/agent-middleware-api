1: """Outbound-URL guard for agent-driven navigation and proxied requests.
2:
3: Agent-supplied URLs reach two genuinely dangerous sinks: the AWI Playwright
4: bridge (a real headless browser that will happily open ``file://`` and
5: intranet URLs) and the behavioral sandbox's HTTP proxy mode (a raw
6: ``aiohttp`` request). Both run inside our infrastructure, so an unchecked
7: URL is an SSRF primitive against link-local metadata services, RFC1918
8: networks, and the host filesystem.
9:
10: ``check_outbound_url`` returns ``None`` when the URL is safe to fetch and a
11: short machine-readable reason string when it must be blocked. Scheme and
12: literal-address checks are unconditional; hostname checks resolve DNS and
13: block names that resolve to non-global addresses. Resolution failure fails
14: closed (``dns_resolution_failed``): the guard cannot vouch for a name it
15: could not resolve, and the fetching client (a headless browser with its own
16: resolver, or a later retry) may well resolve it — possibly to an intranet
17: address. This matches the upstream MCP URL guard. Tests that exercise
18: public hostnames pin ``_resolve_host`` instead of relying on live DNS. This
19: is a pre-flight guard, not a substitute for network-level egress policy,
20: and it intentionally does not try to defeat DNS rebinding.
21:
22: Setting ``ALLOW_PRIVATE_NETWORK_TARGETS=true`` (local development against
23: mock servers) skips only the private-address checks; non-http(s) schemes
24: such as ``file://`` and ``javascript:`` are never allowed.
25: """

41: def _address_blocked(address: str) -> bool:
42:     try:
43:         ip = ipaddress.ip_address(address)
44:     except ValueError:
45:         return False
46:     # is_global is False for loopback, RFC1918/ULA, link-local (including
47:     # 169.254.169.254 metadata), CGNAT shared space, multicast, reserved,
48:     # and unspecified addresses.
49:     return not ip.is_global

52: async def _resolve_host(host: str):
53:     """Resolve a hostname through the event loop; split out for deterministic tests."""
54:
55:     return await asyncio.get_running_loop().getaddrinfo(
56:         host, None, type=socket.SOCK_STREAM
57:     )

60: async def check_outbound_url(url: str) -> str | None:
61:     """Return a block reason for an agent-supplied outbound URL, or None."""
62:     try:
63:         parts = urlsplit(url)
64:     except ValueError:
65:         return "unparseable_url"
66:
67:     if parts.scheme.lower() not in _ALLOWED_SCHEMES:
68:         return "scheme_not_allowed"
69:
70:     host = parts.hostname
71:     if not host:
72:         return "missing_host"
73:
74:     if get_settings().ALLOW_PRIVATE_NETWORK_TARGETS:
75:         return None
76:
77:     host = host.strip("[]").rstrip(".").lower()
78:     if host in _BLOCKED_HOSTNAMES or host.endswith(_BLOCKED_SUFFIXES):
79:         return "private_host_blocked"
80:
81:     if _address_blocked(host):
82:         return "private_address_blocked"
83:
84:     try:
85:         ipaddress.ip_address(host)
86:     except ValueError:
87:         # A hostname, not a literal address: resolve it and require every
88:         # resolved address to be globally routable.
89:         try:
90:             infos = await _resolve_host(host)
91:         except (OSError, UnicodeError):
92:             # socket.gaierror is an OSError; UnicodeError covers names the
93:             # IDNA codec rejects (e.g. an over-long label). Fail closed.
94:             return "dns_resolution_failed"
95:         for info in infos:
96:             if _address_blocked(str(info[4][0])):
97:                 return "private_address_blocked"
98:
99:     return None
