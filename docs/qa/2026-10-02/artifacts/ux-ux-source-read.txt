4: (function () {
5:   "use strict";

7:   var STORAGE_KEY = "amw-a11y";
8:   var MIN_SCALE = 0.9;
9:   var MAX_SCALE = 1.4;
10:   var STEP = 0.1;
11:   var root = document.documentElement;
12: 
13:   var defaults = {
14:     scale: 1,
15:     contrast: false,
16:     motion: false,
17:     spacing: false,
18:     underline: false,
19:   };
20: 
21:   function loadState() {
22:     try {
23:       var raw = window.localStorage.getItem(STORAGE_KEY);
24:       if (!raw) return Object.assign({}, defaults);
25:       var parsed = JSON.parse(raw);
26:       var state = Object.assign({}, defaults, parsed);
27:       state.scale = Math.min(MAX_SCALE, Math.max(MIN_SCALE, Number(state.scale) || 1));
28:       return state;
29:     } catch (error) {
30:       return Object.assign({}, defaults);
31:     }
32:   }
33: 
34:   function saveState(state) {
35:     try {
36:       window.localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
37:     } catch (error) {
38:       /* private mode: preferences simply do not persist */
39:     }
40:   }
41: 
42:   function applyState(state) {
43:     root.style.setProperty("--a11y-scale", String(state.scale));
44:     if (state.contrast) {
45:       root.setAttribute("data-a11y-contrast", "high");
46:     } else {
47:       root.removeAttribute("data-a11y-contrast");
48:     }
49:     if (state.motion) {
50:       root.setAttribute("data-a11y-motion", "reduce");
51:     } else {
52:       root.removeAttribute("data-a11y-motion");
53:     }
54:     if (state.spacing) {
55:       root.setAttribute("data-a11y-spacing", "wide");
56:     } else {
57:       root.removeAttribute("data-a11y-spacing");
58:     }
59:     if (state.underline) {
60:       root.setAttribute("data-a11y-underline", "on");
61:     } else {
62:       root.removeAttribute("data-a11y-underline");
63:     }
64:   }
65: 
66:   var state = loadState();
67:   applyState(state);
68: 
69:   function build() {
70:     if (document.getElementById("amw-a11y-panel")) return;
71:     var toggle = document.createElement("button");
72:     toggle.type = "button";
73:     toggle.className = "amw-a11y-toggle";
74:     toggle.setAttribute("aria-expanded", "false");
75:     toggle.setAttribute("aria-controls", "amw-a11y-panel");
76:     toggle.setAttribute("aria-label", "Accessibility options");
77:     toggle.textContent = "Aa";
78: 
79:     var panel = document.createElement("div");
80:     panel.className = "amw-a11y-panel";
81:     panel.id = "amw-a11y-panel";
82:     panel.setAttribute("role", "dialog");
83:     panel.setAttribute("aria-label", "Accessibility options");
84:     panel.setAttribute("tabindex", "-1");
85:     panel.hidden = true;
86: 
87:     var heading = document.createElement("h2");
88:     heading.textContent = "Accessibility";
89:     panel.appendChild(heading);
90: 
91:     /* Text size row */
92:     var sizeRow = document.createElement("div");
93:     sizeRow.className = "amw-a11y-row";
94:     var sizeLabel = document.createElement("span");
95:     sizeLabel.textContent = "Text size";
96:     sizeLabel.id = "amw-a11y-size-label";
97:     var sizeControls = document.createElement("span");
98:     sizeControls.className = "amw-a11y-size";
99:     sizeControls.setAttribute("role", "group");
100:     sizeControls.setAttribute("aria-labelledby", "amw-a11y-size-label");
101: 
102:     var smaller = document.createElement("button");
103:     smaller.type = "button";
104:     smaller.textContent = "−";
105:     smaller.setAttribute("aria-label", "Decrease text size");
106: 
107:     var readout = document.createElement("output");
108:     readout.setAttribute("aria-live", "polite");
109: 
110:     var larger = document.createElement("button");
111:     larger.type = "button";
112:     larger.textContent = "+";
113:     larger.setAttribute("aria-label", "Increase text size");
114: 
115:     sizeControls.appendChild(smaller);
116:     sizeControls.appendChild(readout);
117:     sizeControls.appendChild(larger);
118:     sizeRow.appendChild(sizeLabel);
119:     sizeRow.appendChild(sizeControls);
120:     panel.appendChild(sizeRow);
121: 
122:     function renderScale() {
123:       readout.value = Math.round(state.scale * 100) + "%";
124:     }
125:     renderScale();
126: 
127:     smaller.addEventListener("click", function () {
128:       state.scale = Math.max(MIN_SCALE, Math.round((state.scale - STEP) * 10) / 10);
129:       applyState(state);
130:       saveState(state);
131:       renderScale();
132:     });
133:     larger.addEventListener("click", function () {
134:       state.scale = Math.min(MAX_SCALE, Math.round((state.scale + STEP) * 10) / 10);
135:       applyState(state);
136:       saveState(state);
137:       renderScale();
138:     });
139: 
140:     /* Switch rows */
141:     var switches = [
142:       { key: "contrast", label: "High contrast" },
143:       { key: "motion", label: "Reduce motion" },
144:       { key: "spacing", label: "Comfortable spacing" },
145:       { key: "underline", label: "Underline links" },
146:     ];
147: 
148:     var switchButtons = {};
149: 
150:     switches.forEach(function (item) {
151:       var row = document.createElement("div");
152:       row.className = "amw-a11y-row";
153:       var label = document.createElement("span");
154:       label.textContent = item.label;
155:       label.id = "amw-a11y-lbl-" + item.key;
156: 
157:       var control = document.createElement("button");
158:       control.type = "button";
159:       control.className = "amw-a11y-switch";
160:       control.id = "amw-a11y-sw-" + item.key;
161:       control.setAttribute("aria-pressed", state[item.key] ? "true" : "false");
162:       control.setAttribute("aria-labelledby", label.id + " " + control.id);
163:       control.textContent = state[item.key] ? "On" : "Off";
164: 
165:       control.addEventListener("click", function () {
166:         state[item.key] = !state[item.key];
167:         control.setAttribute("aria-pressed", state[item.key] ? "true" : "false");
168:         control.textContent = state[item.key] ? "On" : "Off";
169:         applyState(state);
170:         saveState(state);
171:       });
172: 
173:       switchButtons[item.key] = control;
174:       row.appendChild(label);
175:       row.appendChild(control);
176:       panel.appendChild(row);
177:     });
178: 
179:     /* Reset */
180:     var reset = document.createElement("button");
181:     reset.type = "button";
182:     reset.className = "amw-a11y-reset";
183:     reset.textContent = "Reset to defaults";
184:     reset.addEventListener("click", function () {
185:       state = Object.assign({}, defaults);
186:       applyState(state);
187:       saveState(state);
188:       renderScale();
189:       Object.keys(switchButtons).forEach(function (key) {
190:         switchButtons[key].setAttribute("aria-pressed", "false");
191:         switchButtons[key].textContent = "Off";
192:       });
193:     });
194:     panel.appendChild(reset);
195: 
196:     function openPanel() {
197:       panel.hidden = false;
198:       toggle.setAttribute("aria-expanded", "true");
199:       panel.focus();
200:     }
201: 
202:     function closePanel(refocus) {
203:       panel.hidden = true;
204:       toggle.setAttribute("aria-expanded", "false");
205:       if (refocus) toggle.focus();
206:     }
207: 
208:     toggle.addEventListener("click", function () {
209:       if (panel.hidden) {
210:         openPanel();
211:       } else {
212:         closePanel(false);
213:       }
214:     });
215: 
216:     document.addEventListener("keydown", function (event) {
217:       if (event.key === "Escape" && !panel.hidden) {
218:         closePanel(true);
219:       }
220:     });
221: 
222:     document.addEventListener("click", function (event) {
223:       if (panel.hidden) return;
224:       if (panel.contains(event.target) || toggle.contains(event.target)) return;
225:       closePanel(false);
226:     });
227: 
228:     document.body.appendChild(toggle);
229:     document.body.appendChild(panel);
230:   }
231: 
232:   if (document.readyState === "loading") {
233:     document.addEventListener("DOMContentLoaded", build);
234:   } else {
235:     build();
236:   }
237: })();
